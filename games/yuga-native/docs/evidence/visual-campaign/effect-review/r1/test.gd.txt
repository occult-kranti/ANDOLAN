func check(label: String, passed: bool, details: Dictionary={}) -> void:
	checks.append({"case":label,"passed":passed,"evidence":details})
	if not passed:push_error("INDEPENDENT EFFECT REVIEW: "+label)
func effect():
	var node=Effects.new();root.add_child(node);node.set_process(false)
	generation+=1
	return node
func event_at(id: String, kind: String) -> Dictionary:
	for event in templates[id]:

	return {}

func mesh_audit(node: Node) -> Dictionary:
	var result:={"finite":true,"index_bounds":true,"triangles":0,"vertices":0,"meshes":0}
	var pending: Array=[node]
	while not pending.is_empty():
		var current: Node=pending.pop_back();pending.append_array(current.get_children())
		if current is Node3D:result.finite=result.finite and current.transform.is_finite()
		if not current is MeshInstance3D or not current.mesh:continue
		for surface in current.mesh.get_surface_count():
			var arrays: Array=current.mesh.surface_get_arrays(surface)
			var points: PackedVector3Array=arrays[Mesh.ARRAY_VERTEX]
			var indices: PackedInt32Array=arrays[Mesh.ARRAY_INDEX]
			result.meshes+=1;result.vertices+=points.size();result.triangles+=indices.size()/3
			for point in points:result.finite=result.finite and point.is_finite() and point.length()<100.0
			for uv in arrays[Mesh.ARRAY_TEX_UV]:result.finite=result.finite and uv.is_finite()
			for color in arrays[Mesh.ARRAY_COLOR]:result.finite=result.finite and is_finite(color.r) and is_finite(color.g) and is_finite(color.b) and is_finite(color.a)
			for index in indices:result.index_bounds=result.index_bounds and index>=0 and index<points.size()
	return result
func state_signature(fx) -> Dictionary:
	var casts: Array=[]
	for cast in fx._casts.values():
		var arrows: Array=[]
		for arrow in cast.arrows:arrows.append({"visible":arrow.root.visible,"transform":arrow.root.global_transform})
		casts.append({"age":cast.age,"phase":cast.main.get_shader_parameter("phase"),"transform":cast.root.global_transform,"arrows":arrows,"geometry":cast.motif.mesh})
	var impacts: Array=[]
	for impact in fx._impacts:impacts.append({"age":impact.age,"phase":impact.material.get_shader_parameter("phase"),"transform":impact.root.global_transform})
	return {"casts":casts,"impacts":impacts}
