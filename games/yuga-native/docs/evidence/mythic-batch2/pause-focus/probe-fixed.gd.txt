var rows:=[]

func frames(n:int):
 for i in range(n): await process_frame
func sample(label:String):
 var focus:Control=root.gui_get_focus_owner()
 var first:Control=world.pause_actions[0]
 var row={"label":label,"window_size":str(root.size),"logical_size":str(root.get_visible_rect().size),"content_scale_size":str(root.content_scale_size),"scroll":world.pause_scroll.scroll_vertical,"first":str(first.get_global_rect()),"scroll_rect":str(world.pause_scroll.get_global_rect()),"focus":str(focus.get_path()) if focus else "none","visible":world.pause_scroll.get_global_rect().encloses(first.get_global_rect())}

 root.size=Vector2i(960,600)

 await frames(8)
